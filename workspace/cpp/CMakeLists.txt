cmake_minimum_required(VERSION 3.20)
project(pattern_processor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(include)
add_executable(pattern_processor src/pattern_processor_main.cpp src/protocol.cpp src/image_validator.cpp src/image_metadata.cpp src/palette_extractor.cpp src/color_delta.cpp src/channel_mapper.cpp src/grid_converter.cpp src/command_generator.cpp)
enable_testing()
add_executable(image_metadata_tests tests/image_metadata_tests.cpp src/image_metadata.cpp src/image_validator.cpp)
add_executable(palette_extraction_tests tests/palette_extraction_tests.cpp src/palette_extractor.cpp)
add_executable(channel_mapping_tests tests/channel_mapping_tests.cpp src/color_delta.cpp src/channel_mapper.cpp)
add_executable(grid_conversion_tests tests/grid_conversion_tests.cpp src/grid_converter.cpp)
add_executable(command_generation_tests tests/command_generation_tests.cpp src/command_generator.cpp)
add_test(NAME image_metadata_tests COMMAND image_metadata_tests)
add_test(NAME palette_extraction_tests COMMAND palette_extraction_tests)
add_test(NAME channel_mapping_tests COMMAND channel_mapping_tests)
add_test(NAME grid_conversion_tests COMMAND grid_conversion_tests)
add_test(NAME command_generation_tests COMMAND command_generation_tests)
